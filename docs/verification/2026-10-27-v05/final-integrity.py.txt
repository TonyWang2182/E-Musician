"""Run only after the recorded 30 minute soak completes; never launches the app."""
import hashlib
import json
import shutil
import sys
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(root / 'scripts'))
from dev import source_fingerprint, Recorder

if '--run' not in sys.argv:
    result = Recorder('verify-final-integrity', 'Release', 'Verify final soak, source and EXE identity; preserve all 33 user files by SHA256; archive exact report bytes').execute([sys.executable, str(Path(__file__).resolve()), '--run'])
    raise SystemExit(result.returncode)

def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))

def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def write(path, data):
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + '\n', encoding='utf-8')

evidence = root / 'docs/verification/2026-10-27-v05'
record_path = root / 'docs/builds/20261027T014700.122339Z-5599e4a30c.json'
record = read(record_path)
assert record['status'] == 'succeeded' and record['exit_code'] == 0, record['status']
assert record['source_changed_during_command'] is False
fingerprint = source_fingerprint()
assert not fingerprint['errors']
assert fingerprint['digest'] == record['source_fingerprint']['digest'] == record['source_fingerprint_end']['digest']
assert fingerprint['digest'] == '34eb7d59f81aac2e98ca90500e8dcce3bb3ce0c23a5c0f6b6fce59d2188c4676'

report_path = root / 'build/performance/058e5a07e3ef44118990bd20795022ad/performance-report.json'
report = read(report_path)
assert report['status'] == 'passed' and report['actual_device_status'] == 'passed'
assert report['warmup_seconds'] >= 120 and report['actual_measure_seconds'] >= 1800
assert report['transport_stayed_playing'] and report['first_transport_stop'] is None
assert report['xrun_delta'] == 0 and report['b01_acceptance_complete'] is False
assert report['actual_sample_rate'] == 48000 and report['actual_buffer_size'] == 256
assert report['third_party_plugins'] == 0
curated = evidence / '34eb7d59-performance-30min.json'
shutil.copyfile(report_path, curated)
assert sha(report_path) == sha(curated)

baseline_path = root / '.cache/v05-inventory/user-files-before.json'
baseline = read(baseline_path)
details = {}
total_bytes = 0
for relative, expected in baseline.items():
    path = root / relative
    size = path.stat().st_size
    actual = sha(path)
    total_bytes += size
    details[relative] = {'sha256': actual, 'bytes': size, 'unchanged': actual == expected}
assert all(item['unchanged'] for item in details.values())
assert len(details) == 33 and total_bytes == 410250775
write(root / '.cache/v05-inventory/user-files-after-final.json', details)
exe = root / 'build/bin/Release/E-musician.exe'
exe_sha = sha(exe)
assert exe_sha == '1a72c25210e296898167db1e33f57f4cbee0b21fa8c358e636917ee8f23024dd'

samples = report['memory_samples']
memory = {}
for field in ('private_bytes', 'working_set_bytes'):
    values = [s[field] for s in samples]
    memory[field] = {'start': values[0], 'end': values[-1], 'min': min(values), 'max': max(values), 'end_minus_start': values[-1] - values[0]}
result = {
    'verified_at': datetime.now(timezone.utc).isoformat(),
    'source_fingerprint': fingerprint['digest'],
    'source_files_count': len(fingerprint['files']),
    'source_unchanged_during_soak': True,
    'release_exe_sha256': exe_sha,
    'soak_record': '../../builds/' + record_path.name,
    'soak_report': {'file': curated.name, 'sha256': sha(curated), 'bytes': curated.stat().st_size},
    'memory_summary': memory,
    'user_files_protection': {'count': len(details), 'bytes': total_bytes, 'all_sha256_unchanged': True, 'exact_inventory': 'Local ignored .cache/v05-inventory only; excluded from publication'},
    'scope': report['scope'],
    'b01_acceptance_complete': False,
    'b01_missing': report['b01_missing'],
}
write(evidence / '34eb7d59-final-integrity.json', result)
print(json.dumps(result, indent=2))
print(json.dumps({key: value for key, value in report.items() if key != 'memory_samples'}, indent=2))
