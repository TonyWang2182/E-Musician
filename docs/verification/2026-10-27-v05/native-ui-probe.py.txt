"""Owned native UI fixture with file-backed local test commands; quiet playback only when explicitly requested."""
import argparse, hashlib, json, sys, time, uuid
from pathlib import Path

root=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(root/'scripts'))
from agent_client import Client

p=argparse.ArgumentParser();p.add_argument('--exe',required=True,type=Path);p.add_argument('--directory',required=True,type=Path)
a=p.parse_args();directory=a.directory.resolve();directory.relative_to((root/'build').resolve());directory.mkdir(parents=True,exist_ok=False)
report={'scope':'Native UI and Agent control of one owned test process using copies; quiet playback only when requested','events':[],
        'probe_script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'executable_sha256':hashlib.sha256(a.exe.read_bytes()).hexdigest()}
(directory/'ready.json').write_text(json.dumps({'directory':str(directory)}),encoding='utf-8')
print(directory,flush=True)
try:
 with Client(a.exe,90,show=True) as client:
  (directory/'state.json').write_text(json.dumps(client.state(),indent=2),encoding='utf-8')
  for sequence in range(1,1001):
   command=directory/f'command-{sequence}.json';deadline=time.monotonic()+1200
   while not command.exists():
    if time.monotonic()>deadline: raise TimeoutError('No UI probe command for 20 minutes')
    if client.process.poll() is not None: raise RuntimeError('Owned UI process exited unexpectedly')
    time.sleep(.1)
   request=json.loads(command.read_text(encoding='utf-8-sig'))
   if request.get('probe')=='quit':break
   try:
    response=client.request(request) if request.get('request_id') is None else client.raw(request)
    state=client.state();result={'request':request,'response':response,'state':state}
   except Exception as e:result={'request':request,'error':str(e)}
   report['events'].append(result)
   (directory/f'response-{sequence}.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
   (directory/'report.json').write_text(json.dumps(report,indent=2),encoding='utf-8')
  report['closed']=True
finally:
 (directory/'report.json').write_text(json.dumps(report,indent=2),encoding='utf-8')
